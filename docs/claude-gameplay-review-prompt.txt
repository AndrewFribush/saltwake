import type {RunState} from './state';
export type ItemLocation={kind:'floor';room:number;x:number;y:number}|{kind:'bag'|'weapon'|'armor'|'consumed'};
export type RunItem={id:string;type:keyof typeof catalog;origin:'town'|'dungeon';location:ItemLocation};
export type InventoryAction={type:'pickup'|'equip'|'use'|'drop';itemId:string};
export const isConsumable=(type:RunItem['type'])=>type==='tonic'||type==='fineTonic';
export function addItem(s:RunState,item:RunItem){
 if(s.items[item.id])throw Error('Duplicate item ID');
 s.items[item.id]=structuredClone(item);
 (item.origin==='town'?s.originalTownIds:s.reservedItemIds).push(item.id);
}
export function accessible(s:RunState,i:RunItem){const l=i.location;return l.kind!=='consumed'&&(l.kind!=='floor'||(l.room===s.roomIndex&&l.x===s.player.x&&l.y===s.player.y));}
// Called only on a clone. A rejected operation never changes caller state.
export function inventoryAction(s:RunState,a:InventoryAction):boolean{
 const item=s.items[a.itemId];if(!item||!accessible(s,item))return false;
 const bagCount=()=>Object.values(s.items).filter(i=>i.location.kind==='bag').length;
 if(a.type==='use'){
  if(!isConsumable(item.type)||s.player.hp>=12)return false;
  s.player.hp=Math.min(12,s.player.hp+(item.type==='tonic'?4:6));item.location={kind:'consumed'};s.log.push(`Used ${catalog[item.type].name}.`);return true;
 }
 if(a.type==='pickup'){
  if(item.location.kind!=='floor'||bagCount()>=6)return false;item.location={kind:'bag'};
 }else if(a.type==='drop'){
  if(item.origin==='town'||item.location.kind!=='bag')return false;
  item.location={kind:'floor',room:s.roomIndex,x:s.player.x,y:s.player.y};
 }else{
  const slot=item.type==='armor'?'armor':item.type==='club'||item.type==='sword'?'weapon':null;
  if(!slot||item.location.kind===slot)return false;
  const old=Object.values(s.items).find(i=>i.location.kind===slot);
  const room=bagCount()-(item.location.kind==='bag'?1:0)<6;
  if(old&&!room&&old.origin==='town')return false;
  if(old)old.location=room?{kind:'bag'}:{kind:'floor',room:s.roomIndex,x:s.player.x,y:s.player.y};
  item.location={kind:slot};
  if(slot==='weapon')s.player.weapon=item.type as 'club'|'sword';else s.player.armor=true;
 }
 s.log.push(`${a.type}: ${catalog[item.type].name} (${item.origin}).`);return true;
}

export function disposition(s:RunState,outcome:Outcome):Disposition{

export function act(s:RunState,a:Action):RunState {

  if(!atClearedExit(s)||(a.type==='continue'&&!hasNextRoom(s)))return s;

 if(a.type==='attack'&&!s.enemies.some(e=>e.id===a.enemyId&&distance(s.player,e)===1))return s;
 const n=structuredClone(s);

  n.log.push(`You hit ${e.id} for ${damage}.`);n.enemies=n.enemies.filter(e=>e.hp>0);
 }
 n.turn++;resolveEnemies(n,a.type==='guard');return n;
}

import {distance,walkable,type RunState,type Point} from './state';
export const same=(a:Point,b:Point)=>a.x===b.x&&a.y===b.y;
// A mutable turn-local guard is shared by enemies and hazards: only one hit is halved.
export function resolveEnemies(s:RunState,guard:boolean){
 const hitPlayer=(damage:number,source:string)=>{
  const raw=Math.max(1,damage-(s.player.armor?1:0));const hit=guard?Math.ceil(raw/2):raw;guard=false;
  s.player.hp=Math.max(0,s.player.hp-hit);s.log.push(`${source} hits for ${hit}.`);
 };
 for(const e of [...s.enemies].sort((a,b)=>a.id<b.id?-1:a.id>b.id?1:0)){
  if(s.player.hp<=0)break;
  if(e.hp<=0)continue;
  if(e.recovering){e.recovering=false;continue;}
  if(e.kind==='charger'&&e.lane?.length){
   for(const tile of e.lane){
    if(same(tile,s.player)){hitPlayer(4,e.id);break;}
    const other=s.enemies.find(other=>other.id!==e.id&&other.hp>0&&same(other,tile));
    if(other){other.hp-=4;s.log.push(`${e.id} strikes ${other.id} for 4.`);break;}
    e.x=tile.x;e.y=tile.y;
   }
   e.recovering=true;
  }else if(distance(e,e.intent)===1&&same(e.intent,s.player)){
   hitPlayer(3,e.id);
  }else if(walkable(s,e.intent)&&!same(e.intent,s.player)&&!s.enemies.some(other=>other.id!==e.id&&other.hp>0&&same(other,e.intent))){e.x=e.intent.x;e.y=e.intent.y;}
 }
 for(const trap of s.traps){
  if(trap.armed){
   if(same(trap,s.player)&&s.player.hp>0)hitPlayer(3,'Spikes');
   for(const e of s.enemies)if(e.hp>0&&same(trap,e)){e.hp-=3;s.log.push(`Spikes hit ${e.id} for 3.`);}
  }
  trap.armed=!trap.armed;
 }
 s.enemies=s.enemies.filter(e=>e.hp>0);
 if(s.player.hp===0)s.status='died';
 revealIntents(s);s.log=s.log.slice(-8);
}
export function revealIntents(s:RunState){
 const key=(p:Point)=>`${p.x},${p.y}`;
 const neighbors=(p:Point)=>[{x:p.x-1,y:p.y},{x:p.x+1,y:p.y},{x:p.x,y:p.y-1},{x:p.x,y:p.y+1}];
 const distances=new Map<string,number>([[key(s.player),0]]);const queue:Point[]=[s.player];
 for(let i=0;i<queue.length;i++)for(const next of neighbors(queue[i]))if(walkable(s,next)&&!distances.has(key(next))){distances.set(key(next),distances.get(key(queue[i]))!+1);queue.push(next);}
 for(const e of s.enemies){
  e.lane=[];
  if(e.recovering){e.intent={x:e.x,y:e.y};continue;}
  if(e.kind==='charger'&&(e.x===s.player.x||e.y===s.player.y)){
   const dx=Math.sign(s.player.x-e.x),dy=Math.sign(s.player.y-e.y);
   if(dx||dy){let p={x:e.x+dx,y:e.y+dy};while(walkable(s,p)){e.lane.push(p);p={x:p.x+dx,y:p.y+dy};}}
   if(e.lane.some(p=>same(p,s.player))){e.intent=e.lane[0];continue;}e.lane=[];
  }
  const choices=neighbors(e).filter(p=>walkable(s,p)&&distances.has(key(p))&&!s.enemies.some(other=>other.id!==e.id&&same(other,p)));
  choices.sort((a,b)=>distances.get(key(a))!-distances.get(key(b))!);
  e.intent=choices[0]??{x:e.x,y:e.y};
 }
}
