export function act(s:RunState,a:Action):RunState {
 if(s.status!=='active')return s;
 if(a.type==='move'&&(!walkable(s,a)||distance(s.player,a)!==1||s.enemies.some(e=>e.x===a.x&&e.y===a.y)))return s;
 if(a.type==='attack'&&!s.enemies.some(e=>e.id===a.enemyId&&distance(s.player,e)===1))return s;
 const n=structuredClone(s);
 if(a.type==='move'){n.player.x=a.x;n.player.y=a.y;}
 if(a.type==='attack'){
  const e=n.enemies.find(e=>e.id===a.enemyId)!;const damage=n.player.weapon==='sword'?3:2;e.hp-=damage;
  n.log.push(`You hit ${e.id} for ${damage}.`);n.enemies=n.enemies.filter(e=>e.hp>0);
 }
 n.turn++;resolveEnemies(n,a.type==='guard');return n;
}

FILE packages/game/src/combat.ts
import {distance,walkable,type RunState,type Point} from './state';
const same=(a:Point,b:Point)=>a.x===b.x&&a.y===b.y;
// Intents are locked before the player's action. Movement never retargets a hit.
export function resolveEnemies(s:RunState,guard:boolean){
 for(const e of [...s.enemies].sort((a,b)=>a.id.localeCompare(b.id))){
  if(s.player.hp<=0)break;
  if(distance(e,e.intent)===1&&same(e.intent,s.player)){
   const raw=Math.max(1,3-(s.player.armor?1:0));const hit=guard?Math.ceil(raw/2):raw;guard=false;
   s.player.hp=Math.max(0,s.player.hp-hit);s.log.push(`${e.id} hits for ${hit}.`);
  }else if(walkable(s,e.intent)&&!same(e.intent,s.player)&&!s.enemies.some(other=>other.id!==e.id&&same(other,e.intent))){e.x=e.intent.x;e.y=e.intent.y;}
 }
 if(s.player.hp===0)s.status='died';
 revealIntents(s);
 s.log=s.log.slice(-8);
}
export function revealIntents(s:RunState){
 for(const e of s.enemies){
  const choices=[{x:e.x-1,y:e.y},{x:e.x+1,y:e.y},{x:e.x,y:e.y-1},{x:e.x,y:e.y+1}].filter(p=>walkable(s,p));
  choices.sort((a,b)=>distance(a,s.player)-distance(b,s.player));
  e.intent=choices[0]??{x:e.x,y:e.y};
 }
}
