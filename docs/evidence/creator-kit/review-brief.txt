 if(typeof d.title!=='string'||!d.title.trim()||new TextEncoder().encode(d.title).length>80)throw Error('Name must be 1–80 UTF-8 bytes');
 if(typeof d.fee!=='string'||!/^\d{1,7}$/.test(d.fee)||BigInt(d.fee)>1000000n)throw Error('Entry fee must be a whole number from 0 to 1000000');
 if(!Array.isArray(d.floors)||d.floors.length<2||d.floors.length>4||!Array.isArray(d.placements)||d.placements.length!==d.floors.length)throw Error('Expected 2–4 floors with matching encounters');
 const point=(p:{x:number;y:number})=>({x:p.x,y:p.y});
 const floors=d.floors.map((f,i)=>{const errors=validateCustomFloor(f);if(errors.length)throw Error(`Floor ${i+1}: ${errors.join('; ')}`);return {tiles:[...f.tiles],spawn:point(f.spawn),exit:point(f.exit),sockets:f.sockets.map(point)};});
 const placements=d.placements.map(ps=>{if(!Array.isArray(ps)||ps.length!==4)throw Error('Each floor needs four encounter slots');return ps.map(p=>{

 });});

}
