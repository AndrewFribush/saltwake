import type {ItemType} from '../../shared/src/types';
/** Hypothetical closed-edition accounts, never a quote or demand forecast. */
export function businessRehearsal(input:{rewards:(ItemType|null)[];copies:number;fee:number;placement:number;paidAttempts:number;fullClaims:number;stockCost:number|null;otherCost:number}){
 const {rewards,copies,fee,placement,paidAttempts,fullClaims,stockCost,otherCost}=input;
 for(const n of [copies,fee,placement,paidAttempts,fullClaims,otherCost,...(stockCost===null?[]:[stockCost])])if(!Number.isSafeInteger(n)||n<0||n>1_000_000)throw Error('Use whole numbers from 0 to 1,000,000.');
 if(copies<1)throw Error('At least one funded copy is required.');
 if(fullClaims>paidAttempts)throw Error('Full-pool claims cannot exceed paid attempts.');
 const requirements:Partial<Record<ItemType,number>>={};for(const type of rewards)if(type)requirements[type]=(requirements[type]??0)+copies;
 const hasRewards=Object.keys(requirements).length>0;
 if(hasRewards&&fullClaims>copies)throw Error('Claims exceed funded copies. Model replenishment separately.');
 if(!hasRewards&&fullClaims)throw Error('A no-reward creation has no reward claims.');
 const revenue=fee*paidAttempts;
 // Full committed cost is charged once, even when unclaimed goods go to the bank.
 const cost=hasRewards&&stockCost===null?null:placement+(stockCost??0)+otherCost;
 return {requirements,revenue,totalCost:cost,contribution:cost===null?null:revenue-cost,breakEvenAttempts:cost===null?null:cost===0?0:fee===0?null:Math.ceil(cost/fee),remainingCopies:hasRewards?copies-fullClaims:null};
}

import test from 'node:test';import assert from 'node:assert/strict';import {businessRehearsal} from '../src/business-rehearsal';
const scenario={rewards:['tonic','tonic','sword'] as const,copies:2,fee:3,placement:5,paidAttempts:10,fullClaims:1,stockCost:20,otherCost:2};
const base=()=>({...scenario,rewards:[...scenario.rewards]});
test('whole committed stock cost is expensed; leftovers are not a maker refund',()=>{const r=businessRehearsal(base());assert.deepEqual(r.requirements,{tonic:4,sword:2});assert.equal(r.revenue,30);assert.equal(r.totalCost,27);assert.equal(r.contribution,3);assert.equal(r.breakEvenAttempts,9);assert.equal(r.remainingCopies,1);assert.equal(businessRehearsal({...base(),fullClaims:0}).contribution,3);});
test('unknown acquisition cost cannot become profit or reference-price cost',()=>{const r=businessRehearsal({...base(),stockCost:null});assert.equal(r.contribution,null);assert.equal(r.breakEvenAttempts,null);});
test('no rewards, no fees, exhausted stock and more than four lifetime attempts',()=>{assert.equal(businessRehearsal({...base(),fullClaims:2}).remainingCopies,0);assert.equal(businessRehearsal({...base(),fee:0}).breakEvenAttempts,null);const r=businessRehearsal({...base(),rewards:[],fullClaims:0,stockCost:null});assert.equal(r.totalCost,7);assert.equal(r.remainingCopies,null);assert.equal(r.revenue,30);});
