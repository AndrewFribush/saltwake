export function resolveEnemies(s:RunState,guard:boolean){
 const hitPlayer=(damage:number,source:string)=>{
  const raw=Math.max(1,damage-(s.player.armor?1:0));const hit=guard?Math.ceil(raw/2):raw;guard=false;
  s.player.hp=Math.max(0,s.player.hp-hit);s.log.push(`${source} hits for ${hit}.`);
 };
 for(const e of [...s.enemies].sort((a,b)=>a.id<b.id?-1:a.id>b.id?1:0)){
  if(s.player.hp<=0)break;
  if(e.hp<=0)continue;
  if(s.roguelike&&!e.alerted){if(distance(e,s.player)>5)continue;e.alerted=true;}
  if(e.recovering){e.recovering=false;continue;}
  if(e.kind==='charger'&&e.lane?.length){
   for(const tile of e.lane){
    if(same(tile,s.player)){hitPlayer(4,e.id);break;}
    const other=s.enemies.find(other=>other.id!==e.id&&other.hp>0&&same(other,tile));
    if(other){other.hp-=4;s.log.push(`${e.id} strikes ${other.id} for 4.`);break;}
    e.x=tile.x;e.y=tile.y;
   }
   e.recovering=true;
  }else if(distance(e,e.intent)===1&&same(e.intent,s.player)){
   hitPlayer(3,e.id);
  }else if(walkable(s,e.intent)&&!same(e.intent,s.player)&&!s.enemies.some(other=>other.id!==e.id&&other.hp>0&&same(other,e.intent))){e.x=e.intent.x;e.y=e.intent.y;}
 }
 for(const trap of s.traps){
  if(trap.armed){
   if(same(trap,s.player)&&s.player.hp>0)hitPlayer(3,'Spikes');
   for(const e of s.enemies)if(e.hp>0&&same(trap,e)){e.hp-=3;s.log.push(`Spikes hit ${e.id} for 3.`);}
  }
  trap.armed=!trap.armed;
 }
 s.enemies=s.enemies.filter(e=>e.hp>0);
 if(s.player.hp===0)s.status='died';

export function revealIntents(s:RunState){
 const key=(p:Point)=>`${p.x},${p.y}`;
 const neighbors=(p:Point)=>[{x:p.x-1,y:p.y},{x:p.x+1,y:p.y},{x:p.x,y:p.y-1},{x:p.x,y:p.y+1}];
 const distances=new Map<string,number>([[key(s.player),0]]);const queue:Point[]=[s.player];
 for(let i=0;i<queue.length;i++)for(const next of neighbors(queue[i]))if(walkable(s,next)&&!distances.has(key(next))){distances.set(key(next),distances.get(key(queue[i]))!+1);queue.push(next);}
 for(const e of s.enemies){
  e.lane=[];
  if(s.roguelike&&!e.alerted&&distance(e,s.player)>5){e.intent={x:e.x,y:e.y};continue;}
  if(e.recovering){e.intent={x:e.x,y:e.y};continue;}
  if(e.kind==='charger'&&(e.x===s.player.x||e.y===s.player.y)){
   const dx=Math.sign(s.player.x-e.x),dy=Math.sign(s.player.y-e.y);
   if(dx||dy){let p={x:e.x+dx,y:e.y+dy};while(walkable(s,p)){e.lane.push(p);p={x:p.x+dx,y:p.y+dy};}}
   if(e.lane.some(p=>same(p,s.player))){e.intent=e.lane[0];continue;}e.lane=[];
  if(s.roguelike&&!e.alerted&&distance(e,s.player)>5){e.intent={x:e.x,y:e.y};continue;}
  }
  const choices=neighbors(e).filter(p=>walkable(s,p)&&distances.has(key(p))&&!s.enemies.some(other=>other.id!==e.id&&same(other,p)));
  choices.sort((a,b)=>distances.get(key(a))!-distances.get(key(b))!);
  e.intent=choices[0]??{x:e.x,y:e.y};
 }
}

A shoved enemy onto an armed spike tile takes damage through the existing trap loop, with no special case added.
Acceptance: tests for push into open tile, push rejected against wall, push rejected into another enemy, push onto armed spikes applies the existing 3 damage, charger lane is recomputed by the existing `revealIntents` after displacement. Log line format recorded in the test.

### T1-3 Decide and test "stagger" as a tunable, default off (S)
Depends on: T1-2.
Files: `packages/game/src/combat.ts`, `packages/game/src/state.ts`.
Add an optional `rules.shoveStaggers` flag on `RunState` (default false). When true, a shoved enemy gets `recovering=true` and loses its next action. This is the one tactical knob the playtest will turn.
Acceptance: one test per flag value. The default-off path produces byte-identical results to T1-2's tests.

### T1-4 Add the `escape` action (M)
Depends on: T1-1.
Files: `packages/game/src/actions.ts`, `packages/game/src/generation.ts`.
Rule: `escape` is valid when the player stands on the room exit and at least one enemy is alive. Existing `extract` is unchanged and still requires a cleared exit. Resolution order: increment turn, call `resolveEnemies` with guard false, then if `player.hp>0` set `status='extracted'` and log "You escaped with enemies still breathing." If hp reaches zero the existing died path applies.
Acceptance: tests for escape succeeds when the alerted pursuer is two tiles away, escape fails when a charger lane covers the exit and damage kills the player, escape with a guard action on the previous turn does not carry the guard into the escape turn, and escape is rejected when the room is already clear (player must use `extract`). Items held at escape are whatever is in `items` at that moment; no new inventory rule.

### H4-2 Human test script (S)
Depends on: H4-1.
Files: `docs/playtest/loop-v1/script.md`.

Script, about thirty minutes per tester:
1. **Cold start, 3 minutes.** Tester walks the town with no instruction except "find a dungeon door and go in." Observer notes whether they find the gate and whether the fee display was understood.
2. **Run one.** No hints. Observer records outcome, duration, loot value, cause of death if any, and whether the cache room was entered.
3. **Run two.** Observer asks "what did the highlighted tiles mean?" before the run, then records the same fields.
4. **Run three.** Tester is told once that the club can shove and that escape is allowed with enemies alive. Same fields.
5. **Shop stop.** Tester visits a shop with finite stock, buys or declines, observer notes whether price and remaining stock were read correctly.
6. **Maker flip, 5 minutes.** Tester opens the transparency page for the edition they played and answers "would you fund this floor, and at what fee?" Observer records the answer verbatim.
7. **Questionnaire.** Five questions on a 1 to 5 scale: I wanted another run; I understood why I died or survived; the cache felt worth the risk; the shop prices made sense; I understood what the maker earns. One free-text question: what would you change first.

### H4-3 Run the first round with at least four humans (L, human time)
Depends on: H4-2.

Testers are numbered, not named, in the files. The summary tabulates the recorded fields and the questionnaire medians.
Acceptance: four completed session files exist. Agents may format the summary from the raw files but may not fill in any score.

### H4-4 Stop and refine gates (S)
Depends on: H4-3.

- **Cache gate.** If no tester entered the cache in runs two or three, the cache is either invisible or not worth it; Phase 5 addresses placement before rewards.

- **Fun gate.** The round passes only if the median on "I wanted another run" is four or higher across at least four humans and Andrew signs the summary. An agent never marks this gate.

CODEX PLAN
# Gameplay and Economy Implementation Plan — independent Codex draft

> **For agentic workers:** Execute task-by-task using superpowers:subagent-driven-development or superpowers:executing-plans. Keep checkboxes open until their evidence exists. This draft is input to cross-review, not the merged authority.

**Goal:** Make preparing, adventuring, extracting or losing, trading and authoring form a repeatable game people choose to play.

**Architecture:** Deepen the existing deterministic tile engine first in free practice. Promote accepted mechanics through explicit versioned server/contract boundaries, then improve economic choices and maker feedback using authoritative records. Preserve the fixed-window shell.

**Tech Stack:** TypeScript, React/Vite, SQLite authoritative server/outbox, Rust local NEAR sandbox.

**Spec:** `docs/design/economy-gameplay-review/review-decisions.md`, its transparency follow-up, and `../design/dungeon-economy/GAME-DESIGN-v1.2-reviewed.md`. User rules prevail over numerical proposals.

## Global constraints

- Local development/simulation only. No paid calls, testnet, production, deployments, submissions or unsolicited participant messages.
- Maker funds treasure; ordinary completion/death fees go to maker; unconsumed dungeon loot returns to pool on loss; town gear goes to bank; consumed goods remain spent; closure residual goes to bank.
- Preserve finite funded stock, item origins/IDs, admission quotes, reserves, concurrent capacity, void refunds and exactly-once settlement. Four simultaneous runs is not four lifetime admissions.
- No-reward attractions remain supported. Operator is bank, not an unlabeled player-maker. Demo grants/accounts remain disclosed.
- Fixed map and inspector geometry, one screen, no page scrolling; keyboard and click equivalent actions.
- Historical completion counts are not new evidence. Existing telemetry does not establish fun or organic demand. Unknown acquisition cost is not zero.
- No mandatory oil, new token, PvP, permanent stat progression, broad crafting catalog or speculative multiplayer expansion in the first slice.

## Review focus

1. Displacement invalidating committed intent must never silently retarget a hit (P03).
2. Extraction response, death, expiry and duplicate requests must select exactly one terminal result (P04/P10).
3. A restart during promotion must not reinterpret admitted runs under new rules (P10).
4. Reused/gifted/seeded/transformed goods must not create fictitious profit or double-count costs (P13).
5. Human tests must expose confusion, coercive recovery and small-sample bias rather than optimize only success rates (P07/P15/P18).

Each engineering card targets one reviewable deliverable, roughly 30–120 focused minutes after prerequisites; estimates are planning ranges, not deadlines. Split any card that exceeds this into subcards before marking progress. Human sessions are separate scheduled work, not agent time estimates. Correctness tests should fail for the intended missing behavior, then pass after implementation; no tests that merely mirror a CSS change. Commit only task-scoped files when appropriate; preserve existing uncommitted work, never `git add .`, no push or production merge. A second model reviews source when the eventual implementation warrants it; the cross-review here concerns plans only.

## Minimum and later

Minimum convincing local deliverable is P01–P11 plus truthful existing transparency: a human-accepted expedition and working supply/trade cycle. P13 becomes mandatory before any profit claim. P12/P14–P17 establish better economy play, not prerequisites to observing the first run. Later catalog expansion, buy orders, procedural families and social luxuries must earn a place through evidence. No contest-date promise; estimate the next card only after its dependencies are observed.
